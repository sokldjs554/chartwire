function parseProm(text){
  const out=[];for(const line of String(text||'').split('\n')){
    if(!line||line[0]==='#')continue;
    const m=/^([a-zA-Z_:][a-zA-Z0-9_:]*)(\{([^}]*)\})?\s+(\S+)/.exec(line);if(!m)continue;
    const labels={};if(m[3])for(const kv of m[3].match(/[a-zA-Z_][a-zA-Z0-9_]*="(?:[^"\\]|\\.)*"/g)||[]){const i=kv.indexOf('=');labels[kv.slice(0,i)]=kv.slice(i+2,-1)}
    out.push({name:m[1],labels,value:Number(m[4])});
  }return out;
}
function promSum(rows,name,pred){return rows.filter(r=>r.name===name&&(!pred||pred(r.labels))).reduce((a,r)=>a+r.value,0)}
function promHistP95(rows,name){
  const bs=rows.filter(r=>r.name===`${name}_bucket`).map(r=>[r.labels.le==='+Inf'?Infinity:Number(r.labels.le),r.value]).sort((a,b)=>a[0]-b[0]);
  if(!bs.length)return null;const total=bs[bs.length-1][1];if(!total)return null;for(const [le,cum] of bs)if(cum>=total*.95)return le;return null;
}
